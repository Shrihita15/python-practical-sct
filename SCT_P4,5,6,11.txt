#8. Implement a single-layer neural network using backpropagation to learn the AND gate.
import numpy as np

# AND gate input data
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])

# Target outputs for AND gate
T = np.array([0, 0, 0, 1])

# Initialize weights and bias randomly
weights = np.random.rand(2)
bias = np.random.rand()

# Learning rate
lr = 0.2

# Sigmoid activation function
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# Training for 500 epochs
for epoch in range(500):

    # Train using each input-output pair
    for x, t in zip(X, T):

        # Calculate net input
        net = np.dot(x, weights) + bias

        # Calculate neuron output
        out = sigmoid(net)

        # Calculate error
        error = t - out

        # Calculate delta
        delta = error * out * (1 - out)

        # Update weights
        weights += lr * delta * x

        # Update bias
        bias += lr * delta

# Display predictions
print("Predictions")

for x in X:
    # Calculate predicted output
    y = sigmoid(np.dot(x, weights) + bias)

Question 3.
Implement a simple Genetic Algorithm to find the value of x that maximizes the function: f(x)=x2 for 0 ≤ x ≤ 31.

Represent x using a 5-bit binary chromosome and perform selection, crossover, and mutation.

Code:
import random

# Initial population
pop = [
    [random.randint(0, 1) for _ in range(5)]
    for _ in range(6)
]

# Fitness function
def fitness(c):
    x = int(''.join(map(str, c)), 2)
    return x * x


# Genetic Algorithm
for gen in range(5):
    # Sort population by fitness
    pop.sort(key=fitness, reverse=True)

    # Select the best 2 parents
    parents = pop[:2]
    new_pop = parents.copy()

    # Crossover and mutation
    while len(new_pop) < 6:
        p1 = random.choice(parents)
        p2 = random.choice(parents)

        # Crossover
        point = random.randint(1, 4)
        child = p1[:point] + p2[point:]

        # Mutation
        if random.random() < 0.1:
            i = random.randint(0, 4)
            child[i] = 1 - child[i]

        new_pop.append(child)

    pop = new_pop

    # Display best solution
    best = max(pop, key=fitness)
    print(
        "Generation", gen + 1,
        "Best:", best,
        "Fitness:", fitness(best)
    )

# Final solution
best = max(pop, key=fitness)
print("Final Solution:", best)
print("Final Fitness:", fitness(best))

Question 5. Develop a simple optimization system in which a Genetic Algorithm determines the best fan speed for a given room temperature. Use fuzzy logic to classify the temperature as Cold, Comfortable, or Hot, and use the classification as part of the fitness calculation.

Code:
import random

# Input temperature
temp = float(input("Enter temperature: "))

# Fuzzy membership
cold = max(0, min(1, (25 - temp) / 10))
hot = max(0, min(1, (temp - 25) / 10))
comfortable = max(0, min(1, 1 - abs(temp - 25) / 5))

print("Cold:", round(cold, 2))
print("Comfortable:", round(comfortable, 2))
print("Hot:", round(hot, 2))


# Genetic Algorithm
def fitness(speed):
    target = 100 * hot + 50 * comfortable
    return 100 - abs(speed - target)


# Initial population
population = [random.randint(0, 100) for _ in range(8)]

# Run for 5 generations
for generation in range(5):
    population.sort(key=fitness, reverse=True)

    # Select the best 2 parents
    parents = population[:2]
    new_population = parents.copy()

    # Crossover and mutation
    while len(new_population) < 8:
        p1 = random.choice(parents)
        p2 = random.choice(parents)

        child = (p1 + p2) // 2

        # Mutation
        if random.random() < 0.2:
            child += random.randint(-10, 10)

        # Keep fan speed between 0 and 100
        child = max(0, min(100, child))
        new_population.append(child)

    population = new_population

    best = max(population, key=fitness)
    print(
        "Generation", generation + 1,
        "Best:", best,
        "Fitness:", round(fitness(best), 2)
    )

# Final result
best = max(population, key=fitness)

print("Optimized Fan Speed:", best)
print("Fitness:", round(fitness(best), 2))
