CREATE TABLE Registration (
    Student_ID NUMBER REFERENCES Student(Student_ID),
    Course_ID NUMBER REFERENCES Course(Course_ID),
    PRIMARY KEY (Student_ID, Course_ID)
);

SELECT s.Name, c.Course_Name
FROM Student s
JOIN Registration r ON s.Student_ID = r.Student_ID
JOIN Course c ON r.Course_ID = c.Course_ID;

SELECT c.Course_Name, COUNT(r.Student_ID) AS Total
FROM Course c
LEFT JOIN Registration r ON c.Course_ID = r.Course_ID
GROUP BY c.Course_ID, c.Course_Name;
