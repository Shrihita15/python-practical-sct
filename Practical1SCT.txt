#P1 a)Implement a single neuron with given inputs, weights, and bias using step activation
import numpy as np
#inputs
X = np.array([1,2,3])
#Weights
W = np.array([0.4, 0.5, 0.6])
#bias- without bias the neuron might not classify data correctly
b= 0.2
#weighted sum
z= np.dot(X,W)+b
# step activation
Output = 1 if z>0 else 0
print("Output:", Output)
#output: 1


#P1 b)Design a neuron using sigmoid activation and compute output.
import numpy as np
def sigmoid(x):
    return 1/(1+np.exp(-x))
X= np.array([2,1])
W= np.array([0.3, 0.8])
b = -0.5

z= np.dot(X,W)+b
Output = sigmoid(z)
print("Sigmoid Output:", Output)
#Sigmoid Output: 0.7109495026250039


#P1 c) Implement forward propagation for 2-input neuron using ReLU.
import numpy as np
def relu(x):
    return max(0,x)

X= np.array([1.5, -2])
