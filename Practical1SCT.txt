#P1 a)Implement a single neuron with given inputs, weights, and bias using step activation
import numpy as np
#inputs
X = np.array([1,2,3])
#Weights
W = np.array([0.4, 0.5, 0.6])
#bias- without bias the neuron might not classify data correctly
b= 0.2
#weighted sum
z= np.dot(X,W)+b
# step activation
Output = 1 if z>0 else 0
print("Output:", Output)
#output: 1

#P1 c) Implement forward propagation for 2-input neuron using ReLU.
import numpy as np
def relu(x):
    return max(0,x)

#P1 h )Build a simple 2-layer forward propagation manually.
import numpy as np
#Input
X = np.array([1,2])
#layer 1
W1 = np.array([[0.5, 0.2], [0.3,0.8]])
b1 = np.array([0.1, 0.2])
#layer 2
W2 = np.array([0.7, 0.6])
b2 = 0.3
#forward pass
h = np.dot(X, W1)+ b1
h = np.maximum(0,h) # ReLU
Output = np.dot(h, W2) + b2
print("Final Output:", Output)
Final Output: 2.34
