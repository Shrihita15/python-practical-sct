Q1. Load the Employee Salary dataset into Python using Pandas.

import pandas as pd
import numpy as np
from math import ceil

#LOAD THE DATASET
 
data = pd.read_csv("employee_salary_dataset.csv ")
 
print("Dataset Shape:",data.shape)
print("Total Population:",len(data))

Q2. Determine the required sample size using the 95% confidence level and 5% margin of error.

#SAMPLE SIZE DETERMINATION
N = len(data)     #Population Size 
Z = 1.96   #Z-Score for 95% confidence 
p = 0.5     #Expected variability 
E = 0.05    #5% margin of error

#Initial sample size
 n0 = (Z**2 * p* (1-p)) /(E**2)

#Adjusted sample size for finite population
sample_size = n0 / (1 + ((n0 - 1) / N))
#Round upward sample_size = ceil(sample_size)
print("Required Sample Size:",sample_size)


Q3. Perform the following sampling techniques: 
Simple Random Sampling 
Systematic Sampling 
Stratified Sampling (based on Department) 
Cluster Sampling (based on Job Level) 
Convenience Sampling 
Purposive Sampling (Salary > ₹80,000)

#SIMPLE RANDOM SAMPLING
 
random_sample = data.sample( n = sample_size, random_state = 2)
 
print("Sample size:",len(random_sample))
print(random_sample.head())

#SYSTEMATIC SAMPLING
 
interval = N//sample_size
 
np.random.seed(42)
start = np.random.randint(0,interval)
 
systematic_sample = data.iloc[
    start: interval
].head(sample_size)
 
print("Sample size:", len(systematic_sample))
 
print(systematic_sample.head())

#5. STRATIFIED SAMPLING USING GENDER
stratified_parts = []
 
for gender, group in data.groupby("Department"):
    group_sample_size = round(len(group) / len(data) * sample_size)
    selected_group = group.sample(n=group_sample_size, random_state=42)
    stratified_parts.append(selected_group)
 
stratified_sample = pd.concat(stratified_parts, ignore_index=True)
 
print(stratified_sample.head())

#6. Cluster Sampling
 
clusters = data["Department"].dropna().unique()
 
np.random.seed(42)
selected_cluster = np.random.choice(clusters)
 
cluster_sample = data[data["Department"] == selected_cluster]
 
print("Selected cluster:", selected_cluster)
print(cluster_sample.head())
print("Number of employees in selected cluster:", len(cluster_sample))

#7 CONVENIENCE SAMPLING
convenience_sample = data.head(sample_size)
convenience_sample

#8. Purposive Sampling
purposive_sample = data[data["Monthly_Salary"] >= 80000]
purposive_sample
