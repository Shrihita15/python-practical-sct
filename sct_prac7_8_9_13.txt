rho = 0.7 
categories = [] 
for pattern in patterns: 
    assigned = False 
    for i in range(len(categories)):
        prototype = categories[i] 
        # Calculate similarity
        intersection = np.sum(
            pattern * prototype
        ) 
        pattern_size = np.sum(pattern) 
        similarity = intersection / pattern_size 
        print(
            "Pattern:", pattern,
            "Category:", i + 1,
            "Similarity:", round(similarity, 2)
        ) 
        # Vigilance test
        if similarity >= rho: 
            # Update prototype
            categories[i] = (
                prototype * pattern
            ) 
            assigned = True
            break 
    # Create new category
    if not assigned:
        categories.append(pattern.copy()) 
print("\nFinal ART Categories:") 
for i, category in enumerate(categories): 
    print(
        "Category", i + 1,
        ":", category
    )

#q2
import numpy as np
import matplotlib.pyplot as plt
# Input data
X = np.array([
    [1, 1],
    [1, 2],
    [2, 1],
    [4, 4],
    [5, 1],
    [4, 5]
])
# Target values
y = np.array([0, 0, 0, 1, 1, 1])
# Initialize weights and bias
weights = np.zeros(2)
bias = 0
learning_rate = 0.1
epochs = 100
# Step activation function
def step_function(z):
    return 1 if z >= 0 else 0
# Training
for epoch in range(epochs):
    errors = 0
    for xi, target in zip(X, y):
        # Calculate weighted sum
        z = np.dot(xi, weights) + bias
        # Prediction
        prediction = step_function(z)
        # Calculate error
        error = target - prediction
        # Update weights and bias
        weights += learning_rate * error * xi
        bias += learning_rate * error
        if error != 0:
            errors += 1
    print("Epoch:", epoch + 1, "Errors:", errors)
    # Stop if no errors
    if errors == 0:
        print("Perceptron converged.")
        break
print("\nFinal weights:", weights)
print("Final bias:", bias)
# Plot data
plt.scatter(
    X[y == 0, 0],
    X[y == 0, 1],
    label="Class 0"
)
plt.scatter(
    X[y == 1, 0],
    X[y == 1, 1],
    label="Class 1"
)
# Decision boundary:
# w1*x1 + w2*x2 + b = 0
x_values = np.linspace(0, 6, 100)
y_values = -(weights[0] * x_values + bias) / weights[1]
plt.plot(
    x_values,
    y_values,
    label="Decision Boundary"
)
plt.xlabel("X1")
plt.ylabel("X2")
plt.title("Perceptron Decision Boundary")
plt.legend()
plt.grid()
plt.show()

from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score
#XOR data
X = np.array([

    [0,0],

    [0,1],

    [1,0],

    [1,1]

])
y = np.array([0,1,1,0])
 
#Linear classifier on original data
model1 = LogisticRegression()

prediction1 = model1.predict(X)

    X[:,0],

    X[:,0] * X[:,1]

model2 = LogisticRegression()
model2.fit(X_transformed,y)
prediction2 = model2.predict(X_transformed)
accuracy2 = accuracy_score(y,prediction2)

PRACTICAL no. 13

Question 1. Create a neural network model to predict the final marks of students using Study Hours, Attendance Percentage, Assignment Score, and Internal Test Score as input features. Generate a student dataset, train the neural network, evaluate its performance using Mean Squared Error and R² Score, and predict the final marks for a new student.

Code:

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
 
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.neural_network import MLPRegressor
from sklearn.metrics import mean_squared_error, r2_score
 
# ---------------------------------------------------------
# Step 1. Generate Student Dataset
# ---------------------------------------------------------
 
np.random.seed(10)
n = 200
 
study_hours = np.random.uniform(1, 10, n)
attendance = np.random.uniform(50, 100, n)
assignment_score = np.random.uniform(40, 100, n)
internal_score = np.random.uniform(30, 100, n)
 
# Generate final marks using a realistic relationship
 
final_marks = (
    5 * study_hours +
    0.20 * attendance +
    0.30 * assignment_score +
    0.40 * internal_score +
    np.random.normal(0, 5, n)
)
 
# Limit final marks to 0-100
final_marks = np.clip(final_marks, 0, 100)
 
 
# Create DataFrame
data = pd.DataFrame({
      "Study_Hours": study_hours,
    "Attendance_Percentage": attendance,
    "Assignment_Score": assignment_score,
    "Internal_Test_Score": internal_score,
    "Final_Marks": final_marks
})
 
 
# Display first 5 rows
print("First 5 rows of the dataset:")
print(data.head())

# ---------------------------------------------------------
# Step 3. Display basic information
# ---------------------------------------------------------
print("Dataset shape",data.shape)
print("\nDataset Info:")
print(data.info())
print("\nStatistical Summary")
print(data.describe())

# ---------------------------------------------------------
# Step 4. Separate Input Features and Target
# ---------------------------------------------------------
X = data[
    [
        "Study_Hours",
        "Attendance_Percentage",
        "Assignment_Score",
        "Internal_Test_Score"
    ]
]
 
y = data["Final_Marks"]

# ---------------------------------------------------------
# Step 5. Split Dataset into Training and Testing Sets
# ---------------------------------------------------------
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42
)
 
print("\nTraining sample:", len(X_train))
print("Testing sample:", len(X_test))
 
# ---------------------------------------------------------
# Step 6. Standardize input data 
# ---------------------------------------------------------
scaler = StandardScaler()
 
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

# ---------------------------------------------------------
# Step 7. Create Neural Network Model
# ---------------------------------------------------------
model = MLPRegressor(
    hidden_layer_sizes=(10, 8),
    activation="relu",
    solver="adam",
    learning_rate_init=0.01,
    max_iter=1000,
    random_state=42
)
 
model.fit(X_train_scaled, y_train)
print("\nNeural Network Training Completed.")

# ---------------------------------------------------------
# Step 8. Make Predictions on Test Data
# ---------------------------------------------------------
y_pred = model.predict(X_test_scaled)
result = pd.DataFrame({
    "Actual_Marks": y_test.values,
    "Predicted_Marks": y_pred
})
 
print(result.head(10))
 

# ---------------------------------------------------------
# Step 9. Evaluate the Model
# ---------------------------------------------------------
mse = mean_squared_error(y_test, y_pred)
rmse = np.sqrt(mse)
r2 = r2_score(y_test, y_pred)
 
print("Mean Squared Error (MSE):", round(mse,2))
print("Root Mean Squared Error:",round(rmse,2))
print("R² Score:", round(r2,3))
 
# ---------------------------------------------------------
# Step 10. Plot Actual vs Predicted Marks
# ---------------------------------------------------------
 
plt.figure(figsize=(8, 5))
plt.scatter(y_test, y_pred)
 
plt.xlabel("Actual Final Marks")
plt.ylabel("Predicted Final Marks")
plt.title("Actual vs Predicted Final Marks")
 
# Perfect prediction line
plt.plot(
    [y_test.min(), y_test.max()],
    [y_test.min(), y_test.max()],
    linestyle = "--"
)
plt.show()

# ---------------------------------------------------------
# Step 11. Predict Final Marks for a New Student
# --------------------------------------------------------
new_student = pd.DataFrame({
    "Study_Hours": [7],
    "Attendance_Percentage": [90],
    "Assignment_Score": [85],
    "Internal_Test_Score": [80]
})
 
# Scale the new student's features
new_student_scaled = scaler.transform(new_student)
 
# Predict final marks
predicted_marks = model.predict(new_student_scaled)
 
print("\nPredicted Final Marks:",
      round(predicted_marks[0], 2))
