#P3 a) Write a Python program to simulate the AND logic gate using the McCulloch-Pitts neuron model.Display the truth table.
import pandas as pd
def mcp_neuron(inputs, weights, threshold):
    total = sum(i*w for i, w in zip (inputs, weights))
    return 1 if total >= threshold else 0
weights = [1,1]
threshold = 2
truth_table = []
for x1 in [0,1]:
    for x2 in [0,1]:
        output = mcp_neuron([x1, x2], weights, threshold)
        truth_table.append([x1, x2, output])
df = pd.DataFrame(truth_table, columns = ["Input", "Input2", "And Output"])
print(df)

#P3 f) Implement the XOR gate using a network of MCP neurons (AND, OR, NOT).
import pandas as pd
def AND(a,b):
    return int(a+b>=2)
def OR(a,b):
    return int(a+b>=1)
def NOT(x):
    return int(-x>=0)
table=[]
for a in [0,1]:
    for b in[0,1]:
        or1 = OR(a,b)
        and1=AND(a,b)
        not1=NOT(and1)
        xor=AND(or1, not1)
        table.append([a,b,xor])
df = pd.DataFrame(table, columns=["A", "B", "XOR"])
print(df)
