def extract_data(indicator_code):
    url = f"{BASE_URL}/{indicator_code}"
    params = {
        "format": "json",
        "date": "2000:2024",
        "per_page": 100
    }

    try:
        response = requests.get(url, params=params, timeout=30)
        print("Status code:", response.status_code)
        response.raise_for_status()

        data = response.json()

        if not isinstance(data, list) or len(data) < 2:
            print("Unexpected API response:", data)
            return None

        return data

    except Exception as e:
        print("API error:", e)
        return None

# Convert API data into Spark DataFrames
def get_indicator_dataframe(name, code):
    data = extract_data(code)

    if not isinstance(data, list) or len(data) < 2 or not isinstance(data[1], list):
        raise ValueError(
            f"Invalid World Bank API response for {name}. "
            "Check the API response and internet connection."
        )

    records = data[1]
    cleaned_records = []

    for record in records:
        if record.get("value") is not None:
            cleaned_records.append({
                "year": int(record["date"]),
                name: float(record["value"])
            })

    if not cleaned_records:
        raise ValueError(f"No usable records found for {name}.")

    return spark.createDataFrame(cleaned_records)

Question 4. Transform Using Spark SQL 
Combine the three indicators using the year column.

from pyspark.sql.functions import col, round, when

# Join all three DataFrames using year
india_indicators = spark.sql("""
    SELECT p.year, p.population, g.gdp, u.unemployment
    FROM population p
    INNER JOIN gdp g ON p.year = g.year
    INNER JOIN unemployment u ON p.year = u.year
    ORDER BY p.year
""")

india_indicators.show()

# Calculate GDP per person and unemployment category
final_df = (
    india_indicators
    .withColumn(
        "gdp_per_person",
        round(col("gdp") / col("population"), 2)
    )
    .withColumn(
        "unemployment_category",
        when(col("unemployment") >= 7, "High")
        .when(col("unemployment") >= 4, "Medium")
        .otherwise("Low")
    )
)

final_df.show()

# Create temporary view
final_df.createOrReplaceTempView("india_development")

# Perform analysis using SQL
analysis_df = spark.sql("""
    SELECT
        year,
        ROUND(gdp_per_person, 2) AS gdp_per_person,
        unemployment,
        unemployment_category
    FROM india_development
    ORDER BY year
""")

analysis_df.show()

# Save the final data as Parquet
final_df.write.mode("overwrite").parquet(
    f"{CURATED_PATH}/india_development_data"
)

# Read the saved Parquet data

)

output_df.show()
