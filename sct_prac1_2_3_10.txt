#Implement the XOR gate using a network of MCP neurons (AND, OR, NOT).
import pandas as pd
def AND(a, b):
    return int(a + b >= 2)
def OR(a, b):
    return int(a + b >= 1)
def NOT(x):
    return int(-x >= 0)

table = []

for a in [0, 1]:
    for b in [0, 1]:
        or1 = OR(a, b)
        and1 = AND(a, b)
        not1 = NOT(and1)
        xor = AND(or1, not1)
        table.append([a, b, xor])

df = pd.DataFrame(table, columns=["A", "B", "XOR"])
print(df)
