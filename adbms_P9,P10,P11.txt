# Q9.9 Sign a short message with the private key, verify with the public key
message = b"Attendance of Amit Sharma is 85"

signature = private_key.sign(
    message,
    padding.PSS(
        mgf=padding.MGF1(hashes.SHA256()),
        salt_length=padding.PSS.MAX_LENGTH
    ),
    hashes.SHA256()
)

print("\nQ9.9 Digital signature created.")
print("Signature (first 40 bytes):", signature[:40])

try:
    public_key.verify(
        signature,
        message,
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256()),
            salt_length=padding.PSS.MAX_LENGTH
        ),
        hashes.SHA256()
    )
    print("Signature is VALID - message is authentic.")
except InvalidSignature:
    print("Signature is INVALID.")

# tampered message

    public_key.verify(
        signature,
        b"Attendance of Amit Sharma is 99",
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256()),
            salt_length=padding.PSS.MAX_LENGTH
        ),
        hashes.SHA256()
    )
    print("Tampered message: signature VALID")
except InvalidSignature:
    print("Tampered message: signature is INVALID - message was changed.")

# Merkle root function (same logic as your worksheet; if a level has an odd
# number of hashes the last one is repeated, so any number of records works)
def merkle_root(hash_list):
    hash_list = hash_list.copy()

    while len(hash_list) > 1:
        if len(hash_list) % 2 == 1:
            hash_list.append(hash_list[-1])

        new_level = []
        for i in range(0, len(hash_list), 2):
            combined = hash_list[i] + hash_list[i + 1]
            new_level.append(calculate_hash(combined))

        hash_list = new_level

    return hash_list[0]


# Build the full tree (list of levels) - same as your worksheet Q9
def build_merkle_tree(hash_list):
    current_level = hash_list.copy()
    tree = [current_level]

    while len(current_level) > 1:
        if len(current_level) % 2 == 1:
            current_level = current_level + [current_level[-1]]

        new_level = []
        for i in range(0, len(current_level), 2):
            combined = current_level[i] + current_level[i + 1]
            new_level.append(calculate_hash(combined))

        tree.append(new_level)
        current_level = new_level

    return tree

# Q20.10 Merkle proof - verify ONE result without checking every record
def merkle_proof(hash_list, index):
    """Return the sibling hashes needed to rebuild the root for hash_list[index]."""
    proof = []
    level = hash_list.copy()
    while len(level) > 1:
        if len(level) % 2 == 1:
            level.append(level[-1])
        sibling = index ^ 1                                 # neighbour at this level
        proof.append((level[sibling], "left" if sibling < index else "right"))
        level = [calculate_hash(level[i] + level[i + 1]) for i in range(0, len(level), 2)]
        index //= 2
    return proof


def verify_with_proof(record, proof, root):
    h = calculate_hash(record)
    for sibling, side in proof:
        h = calculate_hash(sibling + h) if side == "left" else calculate_hash(h + sibling)
    return h == root

print("""
Explanation:
A Merkle proof contains only the sibling hashes on the path from one record
(leaf) to the root. To verify one result we hash the record, combine it with
each sibling hash step by step, and compare the final hash with the stored
Merkle root. If it matches, the record is authentic. For n records only about
log2(n) hashes are needed (3 for 8 records) instead of comparing all n records.
""")
